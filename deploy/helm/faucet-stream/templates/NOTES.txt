  curl -sX POST localhost:{{ .Values.service.port }}/v1/runs -H "Authorization: Bearer $TOKEN" -H 'Content-Type: application/yaml' --data-binary @pipeline.yaml

{{- if and .Values.serve.templatesSync.enabled (eq .Values.serve.history.backend "memory") }}

WARNING: serve.templatesSync is on with serve.history.backend=memory. Synced
templates, their versions and launches live in the run history, so every pod
restart starts the registry from scratch (each template re-registers as
version 1 and `stable` follows only the sidecars again). Use sqlite (with
serve.persistence) or postgres to keep them.
{{- end }}
{{- if and .Values.serve.triggers.enabled (eq .Values.serve.history.backend "memory") }}

start_at: beginning re-fires for objects it already loaded).
{{- end }}
